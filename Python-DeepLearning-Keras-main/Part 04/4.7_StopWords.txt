from konlpy.tag import Okt

okt = Okt()
okt_tokenize = okt.morphs('첫 번째 이야기는 점들을 연결하는 것입니다.')

stopwords = ['는', '들', '을', '것', '하는', '입니다']

print('\n as-is : ', okt_tokenize)
print('\n to-be : ', [word for word in okt_tokenize if word not in stopwords])
