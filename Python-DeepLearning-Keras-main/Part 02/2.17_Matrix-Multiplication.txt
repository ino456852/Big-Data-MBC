import numpy as np

import numpy as np

a = np.arange(1,5).reshape(2,2) # 행렬
b = np.arange(1,5).reshape(2,2) # 행렬

print(a)
print(b, '\n')

# 곱셉
print( 'multiply', np.multiply(a, b), sep='\n')

# 행렬곱
print( '\n dot', np.dot(a, b), sep='\n')
